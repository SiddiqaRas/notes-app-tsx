CREATE TABLE users(
user_id SERIAL PRIMARY KEY,
user_name VARCHAR(50) NOT NULL,
user_email CITEXT UNIQUE NOT NULL,
password_hash TEXT NOT NULL,
u_role TEXT NOT NULL DEFAULT 'user',
created_at TIMESTAMP DEFAULT NOW()
);
CREATE TABLE projects(
project_id SERIAL PRIMARY KEY,
project_name VARCHAR(50) NOT NULL,
project_description TEXT NOT NULL,
created_by INT REFERENCES users(user_id ON DELETE CASCADE),
created_at TIMESTAMP DEFAULT NOW()
);
CREATE TABLE project_assignments(
aproject_id INT REFERENCES projects(project_id) ON DELETE CASCADE,
auser_id INT REFERENCES users(user_id) ON DELETE CASCADE,
assigned_at TIMESTAMP DEFAULT NOW(),
PRIMARY KEY(aproject_id, auser_id)
);
CREATE TABLE project_updates(
update_id SERIAL PRIMARY KEY,
p_update TEXT NOT NULL,
created_at TIMESTAMP DEFAULT NOW(),
updated_at TIMESTAMP,
update_created_by INT REFERENCES users(user_id),
updated_project_id INT REFERENCES projects(project_id) ON DELETE CASCADE,
last_edited_by INT REFERENCES users(user_id)
);
